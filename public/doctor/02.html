    Repository

    import {DefaultCrudRepository, HasManyRepositoryFactory, repository} from '@loopback/repository';
    import {Country, State} from '../models';
    import {MongodbDataSource} from '../datasources';
    import {inject, Getter} from '@loopback/core';
    import {StateRepository} from './state.repository';
    
    export class CountryRepository extends DefaultCrudRepository<
      Country,
      typeof Country.prototype.id
    > {
      public readonly states: HasManyRepositoryFactory;
    
      constructor(
        @inject('datasources.MongoDB') dataSource: MongodbDataSource,
        @repository.getter('StateRepository')
        protected stateRepositoryGetter: Getter,
      ) {
        super(Country, dataSource);
        this.states = this.createHasManyRepositoryFactoryFor(
          'states',
          stateRepositoryGetter,
        );
        this.registerInclusionResolver('states', this.states.inclusionResolver);
      }
    }
    
    

    import {DefaultCrudRepository, BelongsToAccessor, repository} from '@loopback/repository';
    import {State, Country} from '../models';
    import {MongodbDataSource} from '../datasources';
    import {inject, Getter} from '@loopback/core';
    import {CountryRepository} from './country.repository';
    
    export class StateRepository extends DefaultCrudRepository<
      State,
      typeof State.prototype.id
    > {
      public readonly country: BelongsToAccessor;
    
      constructor(
        @inject('datasources.MongoDB') dataSource: MongodbDataSource,
        @repository.getter('CountryRepository')
        protected countryRepositoryGetter: Getter,
      ) {
        super(State, dataSource);
        this.country = this.createBelongsToAccessorFor(
          'country',
          countryRepositoryGetter,
        );
        this.registerInclusionResolver('country', this.country.inclusionResolver);
      }
    }

    
    
    Controller

    import {get, post, param, requestBody, getModelSchemaRef} from '@loopback/rest';
    import {Country} from '../models';
    import {CountryRepository} from '../repositories';
    import {repository} from '@loopback/repository';

    export class CountryController {
        constructor(
          @repository(CountryRepository)
          public countryRepo: CountryRepository,
        ) {}
      
        @post('/countries')
        async create(@requestBody({content: {'application/json': {schema: getModelSchemaRef(Country)}}}) country: Country) {
          return this.countryRepo.create(country);
        }
      
        @get('/countries', {
          responses: {
            '200': {
              description: 'List of countries with states',
              content: {
                'application/json': {
                  schema: {
                    type: 'array',
                    items: getModelSchemaRef(Country, {includeRelations: true}),
                  },
                },
              },
            },
          },
        })
        async find() {
          return this.countryRepo.find({include: ['states']});
        }
      }
    
      

    import { get, post, param, requestBody, getModelSchemaRef } from '@loopback/rest';
    import {State} from '../models';
    import {StateRepository} from '../repositories';
    import {repository} from '@loopback/repository';

    export class StateController {
        constructor(
          @repository(StateRepository)
          public stateRepo: StateRepository,
        ) {}
      
        @post('/states')
        async create(@requestBody({content: {'application/json': {schema: getModelSchemaRef(State)}}}) state: State) {
          return this.stateRepo.create(state);
        }
      
        @get('/states')
        async find() {
          return this.stateRepo.find({include: ['country']});
        }
    }
    
    
    Unit Testing    

    import {Client, expect} from '@loopback/testlab';
    import {CountryStateApiApplication} from '../../application';
    import {setupApplication} from './test-helper';

    let app: CountryStateApiApplication;
    let client: Client;

    describe('Country-State API', () => {
        before('setupApplication', async () => {
            ({app, client} = await setupApplication());
        });

        after(async () => {
            await app.stop();
        });

        it('creates a country', async () => {
            const country = {name: 'India'};
            const res = await client.post('/countries').send(country).expect(200);
            expect(res.body).to.containDeep(country);
            expect(res.body).to.have.property('id');
        });

        it('creates a state and links to country', async () => {
            const countryRes = await client.post('/countries').send({name: 'USA'});
            const countryId = countryRes.body.id;

            const state = {name: 'California', countryId};
            const stateRes = await client.post('/states').send(state).expect(200);

            expect(stateRes.body).to.containDeep(state);
        });

        it('lists countries with related states', async () => {
            const india = await client.post('/countries').send({name: 'India'});
            const countryId = india.body.id;
            await client.post('/states').send({name: 'West Bengal', countryId});
            await client.post('/states').send({name: 'Gujarat', countryId});

            const res = await client.get('/countries?filter[include][]=states');
            const foundCountry = res.body.find((c: any) => c.id === countryId);
            expect(foundCountry.states).to.be.an.Array();
            expect(foundCountry.states).length(2);
        });

        it('lists states with related country', async () => {
            const res = await client.get('/states?filter[include][]=country');
            for (const state of res.body) {
            expect(state.country).to.be.an.Object();
            expect(state.country.name).to.be.String();
            }
        });
    });
