use actix_web::{web, App, HttpServer, HttpResponse, Responder};
use serde::{Deserialize, Serialize};
use sqlx::mysql::MySqlPool;
use dotenvy::dotenv;
use std::env;

#[derive(Serialize, Deserialize)]
struct Product {
    id: Option,
    name: String,
    price: f32,
}

// CREATE
async fn create_product(pool: web::Data, item: web::Json) -> impl Responder {
    let result = sqlx::query!(
        "INSERT INTO products (name, price) VALUES (?, ?)",
        item.name,
        item.price
    )
    .execute(pool.get_ref())
    .await;

    match result {
        Ok(_) => HttpResponse::Ok().body("Created"),
        Err(e) => HttpResponse::InternalServerError().body(e.to_string()),
    }
}

// READ ALL
async fn get_products(pool: web::Data) -> impl Responder {
    let rows = sqlx::query!("SELECT id, name, price FROM products")
        .fetch_all(pool.get_ref())
        .await;

    match rows {
        Ok(data) => HttpResponse::Ok().json(data),
        Err(e) => HttpResponse::InternalServerError().body(e.to_string()),
    }
}

// READ ONE
async fn get_product(pool: web::Data, id: web::Path) -> impl Responder {
    let row = sqlx::query!(
        "SELECT id, name, price FROM products WHERE id = ?",
        *id
    )
    .fetch_one(pool.get_ref())
    .await;

    match row {
        Ok(data) => HttpResponse::Ok().json(data),
        Err(_) => HttpResponse::NotFound().body("Not found"),
    }
}

// UPDATE
async fn update_product(
    pool: web::Data,
    id: web::Path,
    item: web::Json,
) -> impl Responder {
    let result = sqlx::query!(
        "UPDATE products SET name = ?, price = ? WHERE id = ?",
        item.name,
        item.price,
        *id
    )
    .execute(pool.get_ref())
    .await;

    match result {
        Ok(_) => HttpResponse::Ok().body("Updated"),
        Err(e) => HttpResponse::InternalServerError().body(e.to_string()),
    }
}

// DELETE
async fn delete_product(pool: web::Data, id: web::Path) -> impl Responder {
    let result = sqlx::query!("DELETE FROM products WHERE id = ?", *id)
        .execute(pool.get_ref())
        .await;

    match result {
        Ok(_) => HttpResponse::Ok().body("Deleted"),
        Err(e) => HttpResponse::InternalServerError().body(e.to_string()),
    }
}

#[actix_web::main]
async fn main() -> std::io::Result<()> {
    dotenv().ok();
    let db_url = env::var("DATABASE_URL").expect("DATABASE_URL missing");

    let pool = MySqlPool::connect(&db_url)
        .await
        .expect("DB connection failed");

    HttpServer::new(move || {
        App::new()
            .app_data(web::Data::new(pool.clone()))
            .route("/products", web::post().to(create_product))
            .route("/products", web::get().to(get_products))
            .route("/products/{id}", web::get().to(get_product))
            .route("/products/{id}", web::put().to(update_product))
            .route("/products/{id}", web::delete().to(delete_product))
    })
    .bind(("127.0.0.1", 8080))?
    .run()
    .await
}
