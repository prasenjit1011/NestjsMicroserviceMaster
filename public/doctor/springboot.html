  • Spring Boot Example : Encapsulation, Inheritance, Polymorphism, Abstraction 
    



// Abstraction
abstract class Payment {
    protected double amount;

    // Encapsulation (private data with getter/setter)
    public void setAmount(double amount) {
        this.amount = amount;
    }

    public double getAmount() {
        return amount;
    }

    // Abstract method (must be implemented by child)
    abstract void pay();
}

// Inheritance
class CreditCardPayment extends Payment {

    // Polymorphism (method overriding)
    @Override
    void pay() {
        System.out.println("Paid " + amount + " using Credit Card");
    }
}

class UpiPayment extends Payment {

    // Polymorphism (method overriding)
    @Override
    void pay() {
        System.out.println("Paid " + amount + " using UPI");
    }
}

// Spring Boot Main Application
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;

@SpringBootApplication
public class DemoApplication {

    public static void main(String[] args) {
        SpringApplication.run(DemoApplication.class, args);

        // Polymorphism (same reference, different behavior)
        Payment payment1 = new CreditCardPayment();
        payment1.setAmount(1000);
        payment1.pay();

        Payment payment2 = new UpiPayment();
        payment2.setAmount(500);
        payment2.pay();
    }
}
  • Spring Boot Example : Collections (List, Set, Map, HashMap) 
    
import java.util.*;

public class CollectionExample {

    public static void main(String[] args) {

        // ===== LIST =====
        List list = new ArrayList<>();
        list.add("Java");
        list.add("Spring");
        list.add("Java");

        System.out.println("List: " + list);
        // Output: List: [Java, Spring, Java]


        // ===== SET =====
        Set set = new HashSet<>();
        set.add("Java");
        set.add("Spring");
        set.add("Java");

        System.out.println("Set: " + set);
        // Output: Set: [Java, Spring]  (order may vary)


        // ===== MAP =====
        Map map = new HashMap<>();
        map.put(1, "Java");
        map.put(2, "Spring");
        map.put(1, "Updated Java");

        System.out.println("Map: " + map);
        // Output: Map: {1=Updated Java, 2=Spring}


        // ===== HASHMAP =====
        HashMap hashMap = new HashMap<>();

        hashMap.put("A", 1);
        hashMap.put("B", 2);
        hashMap.put("C", 3);
        hashMap.put("Aa", 100);
        hashMap.put("BB", 200);

        System.out.println("HashMap: " + hashMap);
        // Output: HashMap: {A=1, B=2, C=3, Aa=100, BB=200}
        // (order may vary)

        System.out.println("Get A: " + hashMap.get("A"));
        // Output: Get A: 1


        for (Map.Entry entry : hashMap.entrySet()) {
            System.out.println(entry.getKey() + " -> " + entry.getValue());
            // Output:
            // A -> 1
            // B -> 2
            // C -> 3
            // Aa -> 100
            // BB -> 200
            // (order may vary)
        }
    }
}

  • Multithreading & Concurrency in Java: A Simple Example

    Multithreading = Multiple threads inside a single program running simultaneously.

    
class MyThread extends Thread {
    public void run() {
        System.out.println("Thread is running: " + Thread.currentThread().getName());
    }
}

public class Main {
    public static void main(String[] args) {
        MyThread t1 = new MyThread();
        MyThread t2 = new MyThread();

        t1.start(); // starts thread 1
        t2.start(); // starts thread 2

        System.out.println("Main thread finished");
    }
}
